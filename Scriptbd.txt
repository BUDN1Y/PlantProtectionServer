/****** Object:  Trigger [dbo].[TRG_Equipments_DeletionCheck]    Script Date: 16.10.2026 10:47:36 ******/
SET ANSI_NULLS ON
GO
SET QUOTED_IDENTIFIER ON
GO
create   trigger [dbo].[TRG_Equipments_DeletionCheck]
on [dbo].[Equipments]
instead of delete
as
begin
	set nocount on;
	if exists (
	 select 1
	 from deleted d
	 join [dbo].[MaintenanceSchedule] ms on ms.EquipmentId = d.Id
	 where ISNULL(ms.StatusId, 0) not in (21, 22)
	)
	begin
        raiserror(N'Невозможно удалить оборудование: есть активные записи графика ТО', 16, 1);
        return;
    end;

	 if exists (
	 select 1
	 from deleted d
	 join [dbo].[MaintenanceSchedule] ms on ms.EquipmentId = d.Id
	 join [dbo].[WorkOrders] wo on wo.MaintenanceScheduleId = ms.Id
	 where ISNULL(wo.StatusId, 0) != 16
	)
	 begin
        raiserror(N'Невозможно удалить оборудование: есть активные заявки', 16, 1);
        return;
    end;

	delete wo
    from [dbo].[WorkOrders] wo
    join [dbo].[MaintenanceSchedule] ms on ms.Id = wo.MaintenanceScheduleId
    join deleted d on d.Id = ms.EquipmentId;


    delete ms
    from [dbo].[MaintenanceSchedule] ms
    join deleted d on d.Id = ms.EquipmentId;


    delete e
    from [dbo].[Equipments] e
    join deleted d on d.Id = e.Id;

end;
GO
ALTER TABLE [dbo].[Equipments] ENABLE TRIGGER [TRG_Equipments_DeletionCheck]
GO

create trigger [dbo].[TRG_WorkOrders_CheckStatus]
on [dbo].[WorkOrders]
after insert, update
as
begin
	set nocount on;

	if not exists (select 1 from deleted)
	begin

	update wo
	set wo.StatusId = 11
	from [dbo].[WorkOrders] wo
	join inserted i on i.Id = wo.Id
	return;
	end;

	if exists (
	select 1 
	from inserted i
	join deleted d on i.Id = d.Id
	where ISNULL(i.StatusId, 0) - ISNULL (d.StatusId, 0) not in (0, 1)
	)
	begin
		RAISERROR(N'Статус заявки не может перескакивать ступень заявки', 16, 1);
		ROLLBACK TRANSACTION;
		return;
	end;

end
